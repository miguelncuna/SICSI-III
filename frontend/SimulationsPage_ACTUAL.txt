export function SimulationsPage() {
  const [simulations, setSimulations] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  useEffect(() => {
    let active = true;

    simulationService
      .getAll()
      .then((data) => {
        if (active) {
          setSimulations(Array.isArray(data) ? data : []);
        }
      })
      .catch((err) => {
        if (active) {
          setError(
            err?.message ||
              "Não foi possível carregar as simulações neste momento."
          );
        }
      })
      .finally(() => {
        if (active) {
          setLoading(false);
        }
      });

    return () => {
      active = false;
    };
  }, []);

  const phishingSimulations = useMemo(
    () =>
      simulations.filter(
        (simulation) =>
          simulation?.category?.toLowerCase() === "phishing" ||
          simulation?.id?.startsWith("phishing-")
      ),
    [simulations]
  );

  const completedCount = phishingSimulations.filter(
    (simulation) =>
      simulation?.completed === true ||
      simulation?.status === "completed"
  ).length;

  const progress =
    phishingSimulations.length > 0
      ? Math.round((completedCount / phishingSimulations.length) * 100)
      : 0;

  return (
    <main className="min-h-full bg-slate-50 px-4 py-6 sm:px-6 lg:px-8">
      <div className="mx-auto max-w-7xl">
        {/* =========================================================
            CABEÇALHO
        ========================================================== */}
        <header className="mb-8 flex flex-col gap-5 lg:flex-row lg:items-end lg:justify-between">
          <div>
            <div className="mb-3 flex items-center gap-2 text-xs font-black uppercase tracking-[0.18em] text-blue-600">
              <ShieldAlert size={15} />
              Centro de treino prático
            </div>

            <h1 className="text-4xl font-black tracking-[-0.04em] text-slate-950 sm:text-5xl">
              Simulações
            </h1>

            <p className="mt-3 max-w-2xl text-sm leading-6 text-slate-500 sm:text-base">
              Pratique decisões de segurança digital através de situações
              interactivas inspiradas em ambientes reais.
            </p>
          </div>

          <div className="inline-flex w-fit items-center gap-3 rounded-2xl border border-slate-200 bg-white px-4 py-3 shadow-sm">
            <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-blue-50 text-blue-600">
              <BarChart3 size={19} />
            </div>

            <div>
              <div className="text-[10px] font-black uppercase tracking-[0.16em] text-slate-400">
                Percurso
              </div>
              <div className="mt-0.5 text-sm font-bold text-slate-800">
                5 áreas de treino
              </div>
            </div>
          </div>
        </header>

        {/* =========================================================
            ERRO
        ========================================================== */}
        {error && (
          <div className="mb-6 rounded-2xl border border-red-200 bg-red-50 px-5 py-4 text-sm text-red-700">
            {error}
          </div>
        )}

        {/* =========================================================
            HERO — PHISHING
        ========================================================== */}
        <section className="overflow-hidden rounded-[2rem] border border-slate-200 bg-white shadow-[0_20px_60px_-35px_rgba(15,23,42,0.35)]">
          <div className="grid lg:grid-cols-[1.04fr_0.96fr]">
            {/* Imagem / ambiente */}
            <div className="relative min-h-[440px] overflow-hidden bg-slate-950">
              <img
                src="https://images.unsplash.com/photo-1563986768609-322da13575f3?auto=format&fit=crop&w=1600&q=90"
                alt="Ambiente de trabalho com computador e comunicação digital"
                className="absolute inset-0 h-full w-full object-cover"
              />

              <div className="absolute inset-0 bg-gradient-to-t from-slate-950 via-slate-950/45 to-slate-950/10" />

              <div className="absolute left-6 right-6 top-6 flex items-center justify-between">
                <div className="flex items-center gap-2">
                  <span className="rounded-xl bg-blue-600 px-3 py-2 text-sm font-black text-white shadow-lg">
                    01
                  </span>

                  <span className="rounded-xl border border-white/15 bg-slate-950/70 px-3 py-2 text-[10px] font-black uppercase tracking-[0.16em] text-white backdrop-blur-md">
                    Treino disponível
                  </span>
                </div>

                <div className="flex h-10 w-10 items-center justify-center rounded-xl border border-white/15 bg-slate-950/60 text-red-400 backdrop-blur-md">
                  <ShieldAlert size={19} />
                </div>
              </div>

              <div className="absolute bottom-7 left-7 right-7">
                <div className="mb-3 text-[10px] font-black uppercase tracking-[0.22em] text-blue-300">
                  Laboratório activo
                </div>

                <h2 className="max-w-xl text-3xl font-black leading-tight tracking-[-0.035em] text-white sm:text-4xl">
                  Simulações Reais de Phishing
                </h2>

                <p className="mt-3 max-w-lg text-sm leading-6 text-slate-200">
                  Detecte, analise e decida. Tal como na vida real.
                </p>
              </div>
            </div>

            {/* Informação */}
            <div className="flex flex-col justify-between p-7 sm:p-9 lg:p-10">

                  <Mail size={22} />
                </div>

                <h2 className="text-2xl font-black tracking-tight text-slate-950 sm:text-3xl">
                  Treine como se fosse real.
                </h2>

                      <>
                        cenários

                        interactivos
                      </>
                    }
                  />

                  <StatCard
                    icon={CheckCircle2}

                    label={
                      <>
                        cenários
                        <br />
                        concluídos
                      </>
                    }
                  />

                  <StatCard
                    icon={BarChart3}
                    value={`${progress}%`}
                    label={
                      <>
                        do
                        <br />
                        percurso
                      </>
                    }
                  />
                </div>
              </div>

              <div className="mt-8 flex flex-col gap-3 sm:flex-row">
                <Link
                  to="/student/simulacoes/phishing-01"
                  className="group inline-flex flex-1 items-center justify-center gap-2 rounded-2xl bg-blue-600 px-5 py-4 text-sm font-black text-white shadow-lg shadow-blue-600/20 transition hover:bg-blue-700"
                >
                  Explorar Phishing
                  <ArrowRight
                    size={18}
                    className="transition-transform group-hover:translate-x-1"
                  />
                </Link>

                <Link
                  to="/student/simulacoes/phishing-01"
                  className="inline-flex items-center justify-center gap-2 rounded-2xl border border-slate-200 bg-slate-50 px-5 py-4 text-sm font-bold text-slate-700 transition hover:bg-slate-100"
                >
                  <Info size={17} />
                  Ver detalhes
                </Link>
              </div>
            </div>
          </div>
        </section>

        {/* =========================================================
            ÁREAS DE TREINO
        ========================================================== */}
        <section className="mt-12">
          <div className="mb-6 flex flex-col gap-3 sm:flex-row sm:items-end sm:justify-between">
            <div>
              <div className="text-[10px] font-black uppercase tracking-[0.2em] text-blue-600">
                Áreas de treino
              </div>

              <h2 className="mt-2 text-2xl font-black tracking-tight text-slate-950 sm:text-3xl">
                Escolha o ambiente onde quer praticar.
              </h2>
            </div>

            <div className="text-sm font-medium text-slate-400">
              Disponibilizadas progressivamente
            </div>
          </div>

          {loading ? (
            <div className="grid gap-5 sm:grid-cols-2 xl:grid-cols-4">
              {TRAINING_AREAS.map((area) => (
                <div
                  key={area.number}
                  className="h-[430px] animate-pulse rounded-3xl border border-slate-200 bg-white"
                />
              ))}
            </div>
          ) : (
            <div className="grid gap-5 sm:grid-cols-2 xl:grid-cols-4">
              {TRAINING_AREAS.map((area) => (
                <ComingSoonCard key={area.number} area={area} />
              ))}
            </div>
          )}
        </section>

        {/* =========================================================
            RODAPÉ DA ÁREA
        ========================================================== */}
        <div className="mt-10 flex items-center gap-3 border-t border-slate-200 py-7 text-xs leading-5 text-slate-400">
          <Users size={16} />
          <span>
            Novos ambientes de treino serão disponibilizados
            progressivamente.
          </span>
        </div>
      </div>
    </main>
  );
}

export default SimulationsPage;
